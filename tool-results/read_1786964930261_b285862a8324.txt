   185→// ────────────────────────────────────────
   186→// Custom Tooltip
   187→// ────────────────────────────────────────
   188→
   189→interface ChartTooltipProps {
   190→  active?: boolean;
   191→  payload?: Array<{ name: string; value: number; color?: string }>;
   192→  label?: string;
   193→  formatter?: (value: number) => string;
   194→}
   195→
   196→function ChartTooltip({ active, payload, label, formatter }: ChartTooltipProps) {
   197→  if (!active || !payload?.length) return null;
   198→  return (
   199→    <div className="bg-white border border-slate-200 rounded-lg shadow-lg p-3 text-sm">
   200→      <p className="font-medium text-slate-700 mb-1">{label}</p>
   201→      {payload.map((entry, idx) => (
   202→        <p key={idx} className="text-slate-600" style={{ color: entry.color || '#334155' }}>
   203→          {entry.name}: {formatter ? formatter(entry.value) : entry.value}
   204→        </p>
   205→      ))}
   206→    </div>
   207→  );
   208→}
