   313→function authFetch(url: string, options: RequestInit = {}) {
   314→  const token = typeof window !== 'undefined' ? localStorage.getItem('rtr_token') : null;
   315→  return fetch(url, {
   316→    ...options,
   317→    headers: {
   318→      'Content-Type': 'application/json',
   319→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
   320→      ...options.headers,
   321→    },
   322→  });
   323→}

   329→function LoginScreen({ onLogin }: { onLogin: (user: UserSession, token: string) => void }) {
   330→  const [email, setEmail] = useState('');
   331→  const [password, setPassword] = useState('');
   332→  const [loading, setLoading] = useState(false);
   333→
   334→  const handleLogin = async (e: React.FormEvent) => {
   335→    e.preventDefault();
   336→    if (!email || !password) {
   337→      toast.error('Please enter email and password');
   338→      return;
   339→    }
   340→    setLoading(true);
   341→    try {
   342→      const res = await fetch('/api/auth/login', {
   343→        method: 'POST',
   344→        headers: { 'Content-Type': 'application/json' },
   345→        body: JSON.stringify({ email, password }),
   346→      });
   347→      const data = await res.json();
   348→      if (!res.ok) {
   349→        toast.error(data.error || 'Login failed');
   350→        return;
   351→      }
   352→      localStorage.setItem('rtr_token', data.token);
   353→      localStorage.setItem('rtr_user', JSON.stringify(data.user));
   354→      onLogin(data.user, data.token);
   355→      toast.success('Welcome back!');
   356→    } catch {
   357→      toast.error('Network error. Please try again.');

   359→      setLoading(false);

   556→
   557→function DashboardView() {
   558→  const [stats, setStats] = useState<DashboardStats | null>(null);
   559→  const [leads, setLeads] = useState<Lead[]>([]);
   560→  const [alerts, setAlerts] = useState<Alert[]>([]);
   561→  const [loading, setLoading] = useState(true);
   562→
   563→  useEffect(() => {
   564→    const fetchData = async () => {
   565→      try {
   566→        const [statsRes, leadsRes, alertsRes] = await Promise.all([
   567→          authFetch('/api/dashboard/stats'),
   568→          authFetch('/api/leads?limit=5'),
   569→          authFetch('/api/dashboard/alerts?limit=5'),
   570→        ]);
   571→        const statsData = await statsRes.json();
   572→        if (statsRes.ok) setStats(statsData);
   573→
   574→        const leadsData = await leadsRes.json();
   575→        if (leadsRes.ok) setLeads(leadsData.leads || []);
   576→
   577→        if (alertsRes.ok) {
   578→          const alertsData = await alertsRes.json();
   579→          setAlerts(alertsData.alerts || []);
   580→        }
   581→      } catch {
   582→        // Silently fail for alerts — the endpoint may not exist
   583→      } finally {
   584→        setLoading(false);
   585→      }
   586→    };
   587→    fetchData();
   588→  }, []);
   589→
   590→  const kpiCards = stats
   591→    ? [
   592→        { icon: Truck, label: 'Total Vehicles', value: stats.totalVehicles, color: 'bg-emerald-100 text-emerald-600' },
   593→        { icon: Activity, label: 'Active Vehicles', value: stats.activeVehicles, color: 'bg-green-100 text-green-600' },
   594→        { icon: Users, label: 'Total Drivers', value: stats.totalDrivers, color: 'bg-teal-100 text-teal-600' },
