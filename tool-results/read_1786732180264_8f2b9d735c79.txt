   347→   347→function authFetch(url: string, options: RequestInit = {}) {
   348→   348→  const token = typeof window !== 'undefined' ? localStorage.getItem('rtr_token') : null;
   349→   349→  return fetch(url, {
   350→   350→    ...options,
   351→   351→    headers: {
   352→   352→      'Content-Type': 'application/json',
   353→   353→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
   354→   354→      ...options.headers,
   355→   355→    },
   356→   356→  });
   357→   357→}
