    56→const authFetch = async (url: string, options: RequestInit = {}) => {
    57→  const token = localStorage.getItem('rtr_token');
    58→  const res = await fetch(url, {
    59→    ...options,
    60→    headers: {
    61→      'Content-Type': 'application/json',
    62→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
    63→      ...options.headers,
    64→    },
    65→  });
    66→  if (!res.ok) {
    67→    throw new Error(`Request failed: ${res.status}`);
    68→  }
    69→  return res.json();

   198→// ────────────────────────────────────────
   199→// Custom Tooltip
   200→// ────────────────────────────────────────
   201→
   202→interface ChartTooltipProps {
   203→  active?: boolean;
   204→  payload?: Array<{ name: string; value: number; color?: string }>;
   205→  label?: string;
   206→  formatter?: (value: number) => string;
   207→}
   208→
   209→function ChartTooltip({ active, payload, label, formatter }: ChartTooltipProps) {
   210→  if (!active || !payload?.length) return null;
   211→  return (
   212→    <div className="bg-white border border-slate-200 rounded-lg shadow-lg p-3 text-sm">
   213→      <p className="font-medium text-slate-700 mb-1">{label}</p>
   214→      {payload.map((entry, idx) => (
   215→        <p key={idx} className="text-slate-600" style={{ color: entry.color || '#334155' }}>
   216→          {entry.name}: {formatter ? formatter(entry.value) : entry.value}
   217→        </p>
   218→      ))}
   219→    </div>
   220→  );
   221→}
