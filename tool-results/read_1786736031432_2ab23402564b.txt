   367→function authFetch(url: string, options: RequestInit = {}) {
   368→  const token = typeof window !== 'undefined' ? localStorage.getItem('rtr_token') : null;
   369→  return fetch(url, {
   370→    ...options,
   371→    headers: {
   372→      'Content-Type': 'application/json',
   373→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
   374→      ...options.headers,
   375→    },
   376→  });
   377→}

   383→function LoginScreen({ onLogin }: { onLogin: (user: UserSession, token: string) => void }) {
   384→  const [email, setEmail] = useState('');
   385→  const [password, setPassword] = useState('');
   386→  const [loading, setLoading] = useState(false);
   387→
   388→  const handleLogin = async (e: React.FormEvent) => {
   389→    e.preventDefault();
   390→    if (!email || !password) {
   391→      toast.error('Please enter email and password');
   392→      return;
   393→    }
   394→    setLoading(true);
   395→    try {
   396→      const res = await fetch('/api/auth/login', {
   397→        method: 'POST',
   398→        headers: { 'Content-Type': 'application/json' },
   399→        body: JSON.stringify({ email, password }),
   400→      });
   401→      const data = await res.json();
   402→      if (!res.ok) {
   403→        toast.error(data.error || 'Login failed');
   404→        return;
   405→      }
   406→      localStorage.setItem('rtr_token', data.token);
   407→      localStorage.setItem('rtr_user', JSON.stringify(data.user));
   408→      onLogin(data.user, data.token);
   409→      toast.success('Welcome back!');
   410→    } catch {
   411→      toast.error('Network error. Please try again.');
   412→    } finally {
   413→      setLoading(false);
   414→    }
