     1→     1→'use client';
     2→     2→
     3→     3→import { useState, useEffect, useCallback } from 'react';
     4→     4→import { toast } from 'sonner';
     5→     5→import { motion } from 'framer-motion';
     6→     6→import {
     7→     7→  Plus,
     8→     8→  Search,
     9→     9→  ChevronLeft,
    10→    10→  ChevronRight,
    11→    11→  Eye,
    12→    12→  Send,
    13→    13→  FileText,
    14→    14→  CheckCircle,
    15→    15→  XCircle,
    16→    16→  Clock,
    17→    17→} from 'lucide-react';
    18→    18→import { Button } from '@/components/ui/button';
    19→    19→import { Card, CardContent } from '@/components/ui/card';
    20→    20→import { Badge } from '@/components/ui/badge';
    21→    21→import {
    22→    22→  Dialog,
    23→    23→  DialogContent,
    24→    24→  DialogHeader,
    25→    25→  DialogTitle,
    26→    26→  DialogFooter,
    27→    27→} from '@/components/ui/dialog';
    28→    28→import { Input } from '@/components/ui/input';
    29→    29→import { Label } from '@/components/ui/label';
    30→    30→import { Skeleton } from '@/components/ui/skeleton';
    31→    31→import { Textarea } from '@/components/ui/textarea';
    32→    32→import {
    33→    33→  Select,
    34→    34→  SelectContent,
    35→    35→  SelectItem,
    36→    36→  SelectTrigger,
    37→    37→  SelectValue,
    38→    38→} from '@/components/ui/select';
    39→    39→import {
    40→    40→  Table,
    41→    41→  TableBody,
    42→    42→  TableCell,
    43→    43→  TableHead,
    44→    44→  TableHeader,
    45→    45→  TableRow,
    46→    46→} from '@/components/ui/table';
    47→    47→import { authFetch, formatAED, formatDate, formatDateTime } from '@/lib/api';
    48→    48→import { STATUS_COLORS, QUOTATION_STATUSES, DEFAULT_QUOTATION_ITEMS, QUOTATION_TERMS } from '@/lib/constants';
    49→    49→import type { Quotation, QuotationItem } from '@/lib/types';
    50→    50→
    51→    51→export default function QuotationsView() {
    52→    52→  const [quotations, setQuotations] = useState<Quotation[]>([]);
    53→    53→  const [totalPages, setTotalPages] = useState(1);
    54→    54→  const [total, setTotal] = useState(0);
    55→    55→  const [page, setPage] = useState(1);
    56→    56→  const [search, setSearch] = useState('');
    57→    57→  const [statusFilter, setStatusFilter] = useState<string>('all');
    58→    58→  const [loading, setLoading] = useState(true);
    59→    59→  const [selectedQuotation, setSelectedQuotation] = useState<Quotation | null>(null);
    60→    60→  const [detailOpen, setDetailOpen] = useState(false);
    61→    61→
    62→    62→  // Create quotation
    63→    63→  const [createOpen, setCreateOpen] = useState(false);
    64→    64→  const [submitting, setSubmitting] = useState(false);
    65→    65→  const [items, setItems] = useState<QuotationItem[]>([...DEFAULT_QUOTATION_ITEMS]);
    66→    66→  const [leadSearch, setLeadSearch] = useState('');
    67→    67→  const [leadResults, setLeadResults] = useState<Array<{ id: string; name: string; company: string | null }>>([]);
    68→    68→  const [selectedLeadId, setSelectedLeadId] = useState<string | null>(null);
    69→    69→  const [qNotes, setQNotes] = useState('');
    70→    70→
    71→    71→  const fetchQuotations = useCallback(async () => {
    72→    72→    setLoading(true);
    73→    73→    try {
    74→    74→      const params = new URLSearchParams({ page: String(page), limit: '15' });
    75→    75→      if (search) params.set('search', search);
    76→    76→      if (statusFilter !== 'all') params.set('status', statusFilter);
    77→    77→      const res = await authFetch(`/api/quotations?${params.toString()}`);
    78→    78→      const data = await res.json();
    79→    79→      if (res.ok) {
    80→    80→        setQuotations(data.quotations || []);
    81→    81→        setTotalPages(data.pagination?.totalPages || 1);
    82→    82→        setTotal(data.pagination?.total || 0);
    83→    83→      }
    84→    84→    } catch { toast.error('Failed to load quotations'); }
    85→    85→    finally { setLoading(false); }
    86→    86→  }, [page, search, statusFilter]);
    87→    87→
    88→    88→  useEffect(() => { fetchQuotations(); }, [fetchQuotations]);
    89→    89→
    90→    90→  // Search leads for quotation creation
    91→    91→  useEffect(() => {
    92→    92→    if (!leadSearch.trim()) { setLeadResults([]); return; }
    93→    93→    const timer = setTimeout(async () => {
    94→    94→      try {
    95→    95→        const res = await authFetch(`/api/leads?search=${encodeURIComponent(leadSearch)}&limit=5`);
    96→    96→        const data = await res.json();
    97→    97→        if (res.ok) setLeadResults(data.leads || []);
    98→    98→      } catch { /* ignore */ }
    99→    99→    }, 300);
   100→   100→    return () => clearTimeout(timer);
   101→   101→  }, [leadSearch]);
   102→   102→
   103→   103→  const fetchQuotationDetail = async (id: string) => {
   104→   104→    try {
   105→   105→      const res = await authFetch(`/api/quotations/${id}`);
   106→   106→      const data = await res.json();
   107→   107→      if (res.ok) {
   108→   108→        setSelectedQuotation(data.quotation);
   109→   109→        setDetailOpen(true);
   110→   110→      }
   111→   111→    } catch { toast.error('Failed to load quotation'); }
   112→   112→  };
   113→   113→
   114→   114→  const handleStatusUpdate = async (id: string, status: string) => {
   115→   115→    try {
   116→   116→      const res = await authFetch(`/api/quotations/${id}`, {
   117→   117→        method: 'PATCH',
   118→   118→        body: JSON.stringify({ status }),
   119→   119→      });
   120→   120→      if (res.ok) {
   121→   121→        toast.success(`Quotation marked as ${status}`);
   122→   122→        fetchQuotations();
   123→   123→      }
   124→   124→    } catch { toast.error('Failed to update'); }
   125→   125→  };
   126→   126→
   127→   127→  const handleCreate = async () => {
   128→   128→    if (items.length === 0) { toast.error('Add at least one item'); return; }
   129→   129→    setSubmitting(true);
   130→   130→    try {
   131→   131→      const res = await authFetch('/api/quotations', {
   132→   132→        method: 'POST',
   133→   133→        body: JSON.stringify({
   134→   134→          leadId: selectedLeadId,
   135→   135→          items,
   136→   136→          notes: qNotes || undefined,
   137→   137→          terms: QUOTATION_TERMS,
   138→   138→        }),
   139→   139→      });
   140→   140→      const data = await res.json();
   141→   141→      if (res.ok) {
   142→   142→        toast.success(`Quotation ${data.quotation.quotationNumber} created!`);
   143→   143→        setCreateOpen(false);
   144→   144→        setItems([...DEFAULT_QUOTATION_ITEMS]);
   145→   145→        setQNotes('');
   146→   146→        setSelectedLeadId(null);
   147→   147→        setLeadSearch('');
   148→   148→        fetchQuotations();
   149→   149→      } else { toast.error(data.error || 'Failed to create'); }
   150→   150→    } catch { toast.error('Failed to create quotation'); }
   151→   151→    finally { setSubmitting(false); }
   152→   152→  };
   153→   153→
   154→   154→  const subtotal = items.reduce((s, i) => s + (i.quantity * i.unitPrice), 0);
   155→   155→  const tax = Math.round(subtotal * 5 / 100 * 100) / 100;
   156→   156→  const totalVal = Math.round((subtotal + tax) * 100) / 100;
   157→   157→
   158→   158→  const statusIcon = (status: string) => {
   159→   159→    switch (status) {
   160→   160→      case 'draft': return <FileText className="w-3.5 h-3.5" />;
   161→   161→      case 'sent': return <Send className="w-3.5 h-3.5" />;
   162→   162→      case 'accepted': return <CheckCircle className="w-3.5 h-3.5" />;
   163→   163→      case 'rejected': return <XCircle className="w-3.5 h-3.5" />;
   164→   164→      case 'expired': return <Clock className="w-3.5 h-3.5" />;
   165→   165→      default: return null;
   166→   166→    }
   167→   167→  };
   168→   168→
   169→   169→  // Parse items from JSON string
   170→   170→  const parseItems = (itemsJson: string): QuotationItem[] => {
   171→   171→    try { return JSON.parse(itemsJson); } catch { return []; }
   172→   172→  };
   173→   173→
   174→   174→  return (
   175→   175→    <div className="space-y-5">
   176→   176→      {/* Header */}
   177→   177→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   178→   178→        <div className="flex items-center gap-3">
   179→   179→          <div>
   180→   180→            <h2 className="text-2xl font-bold text-slate-900">Quotations</h2>
   181→   181→            <p className="text-sm text-slate-500">Create and manage customer quotations</p>
   182→   182→          </div>
   183→   183→          <Badge variant="secondary" className="bg-emerald-100 text-emerald-700 border-0 text-sm px-2.5">
   184→   184→            {total}
   185→   185→          </Badge>
   186→   186→        </div>
   187→   187→        <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   188→   188→          <DialogTrigger asChild>
   189→   189→            <Button className="bg-emerald-600 hover:bg-emerald-700 text-white gap-2">
   190→   190→              <Plus className="w-4 h-4" /> New Quotation
   191→   191→            </Button>
   192→   192→          </DialogTrigger>
   193→   193→          <DialogContent className="max-w-2xl max-h-[85vh] overflow-y-auto">
   194→   194→            <DialogHeader><DialogTitle>Create New Quotation</DialogTitle></DialogHeader>
   195→   195→            <div className="space-y-4 py-2">
   196→   196→              {/* Lead Search */}
   197→   197→              <div className="space-y-2">
   198→   198→                <Label>Link to Lead (optional)</Label>
   199→   199→                <Input
   200→   200→                  placeholder="Search lead by name or company..."
   201→   201→                  value={leadSearch}
   202→   202→                  onChange={(e) => setLeadSearch(e.target.value)}
   203→   203→                />
   204→   204→                {leadResults.length > 0 && !selectedLeadId && (
   205→   205→                  <div className="border border-slate-200 rounded-lg max-h-32 overflow-y-auto">
   206→   206→                    {leadResults.map((l) => (
   207→   207→                      <button
   208→   208→                        key={l.id}
   209→   209→                        className="w-full text-left px-3 py-2 hover:bg-slate-50 text-sm flex items-center justify-between"
   210→   210→                        onClick={() => { setSelectedLeadId(l.id); setLeadSearch(l.name + (l.company ? ` - ${l.company}` : '')); setLeadResults([]); }}
   211→   211→                      >
   212→   212→                        <span className="font-medium text-slate-800">{l.name}</span>
   213→   213→                        <span className="text-xs text-slate-400">{l.company || ''}</span>
   214→   214→                      </button>
   215→   215→                    ))}
   216→   216→                  </div>
   217→   217→                )}
   218→   218→                {selectedLeadId && (
   219→   219→                  <div className="flex items-center gap-2 text-sm text-emerald-600">
   220→   220→                    <CheckCircle className="w-4 h-4" />
   221→   221→                    <span>Linked to: {leadSearch}</span>
   222→   222→                    <button className="text-slate-400 hover:text-red-500 ml-auto" onClick={() => { setSelectedLeadId(null); setLeadSearch(''); }}>
   223→   223→                      ✕
   224→   224→                    </button>
   225→   225→                  </div>
   226→   226→                )}
   227→   227→              </div>
   228→   228→
   229→   229→              {/* Line Items */}
   230→   230→              <div>
   231→   231→                <Label className="text-sm font-semibold">Line Items</Label>
   232→   232→                <div className="mt-2 space-y-2">
   233→   233→                  {items.map((item, idx) => (
   234→   234→                    <div key={idx} className="flex items-center gap-2">
   235→   235→                      <Input className="flex-1 h-9 text-sm" placeholder="Description" value={item.description}
   236→   236→                        onChange={(e) => { const u = [...items]; u[idx] = { ...u[idx], description: e.target.value }; setItems(u); }} />
   237→   237→                      <Input className="w-20 h-9 text-sm text-center" type="number" placeholder="Qty" value={item.quantity}
   238→   238→                        onChange={(e) => { const u = [...items]; u[idx] = { ...u[idx], quantity: parseInt(e.target.value) || 0 }; setItems(u); }} />
   239→   239→                      <Input className="w-28 h-9 text-sm text-right" type="number" placeholder="AED" value={item.unitPrice}
   240→   240→                        onChange={(e) => { const u = [...items]; u[idx] = { ...u[idx], unitPrice: parseFloat(e.target.value) || 0 }; setItems(u); }} />
   241→   241→                      <p className="text-sm font-medium text-slate-700 w-24 text-right">
   242→   242→                        {formatAED(item.quantity * item.unitPrice)}
   243→   243→                      </p>
   244→   244→                      <button className="p-1.5 text-red-400 hover:text-red-600 hover:bg-red-50 rounded-lg transition-colors"
   245→   245→                        onClick={() => setItems(items.filter((_, i) => i !== idx))}>✕</button>
   246→   246→                    </div>
   247→   247→                  ))}
   248→   248→                </div>
   249→   249→                <Button variant="outline" size="sm" className="mt-2 gap-1.5"
   250→   250→                  onClick={() => setItems([...items, { description: '', quantity: 1, unitPrice: 0 }])}>
   251→   251→                  <Plus className="w-3.5 h-3.5" /> Add Item
   252→   252→                </Button>
   253→   253→              </div>
   254→   254→
   255→   255→              <div className="space-y-2">
   256→   256→                <Label>Notes</Label>
   257→   257→                <Textarea value={qNotes} onChange={(e) => setQNotes(e.target.value)} placeholder="Additional notes..." rows={2} />
   258→   258→              </div>
   259→   259→
   260→   260→              {/* Totals */}
   261→   261→              <div className="p-4 bg-slate-50 rounded-xl space-y-2">
   262→   262→                <div className="flex justify-between text-sm">
   263→   263→                  <span className="text-slate-500">Subtotal</span>
   264→   264→                  <span className="font-medium">{formatAED(subtotal)}</span>
   265→   265→                </div>
   266→   266→                <div className="flex justify-between text-sm">
   267→   267→                  <span className="text-slate-500">VAT (5%)</span>
   268→   268→                  <span className="font-medium">{formatAED(tax)}</span>
   269→   269→                </div>
   270→   270→                <div className="flex justify-between text-base font-bold border-t border-slate-200 pt-2">
   271→   271→                  <span>Total</span>
   272→   272→                  <span className="text-emerald-700">{formatAED(totalVal)}</span>
   273→   273→                </div>
   274→   274→              </div>
   275→   275→            </div>
   276→   276→            <DialogFooter>
   277→   277→              <Button variant="outline" onClick={() => setCreateOpen(false)}>Cancel</Button>
   278→   278→              <Button className="bg-emerald-600 hover:bg-emerald-700 text-white" onClick={handleCreate} disabled={submitting}>
   279→   279→                {submitting ? 'Creating...' : 'Create Quotation'}
   280→   280→              </Button>
   281→   281→            </DialogFooter>
   282→   282→          </DialogContent>
   283→   283→        </Dialog>
   284→   284→      </div>
   285→   285→
   286→   286→      {/* Filters */}
   287→   287→      <div className="flex flex-col sm:flex-row gap-3">
   288→   288→        <div className="relative flex-1">
   289→   289→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
   290→   290→          <Input placeholder="Search quotation number..." value={search}
   291→   291→            onChange={(e) => { setSearch(e.target.value); setPage(1); }} className="pl-10" />
   292→   292→        </div>
   293→   293→        <Select value={statusFilter} onValueChange={(v) => { setStatusFilter(v); setPage(1); }}>
   294→   294→          <SelectTrigger className="w-full sm:w-44"><SelectValue placeholder="All Status" /></SelectTrigger>
   295→   295→          <SelectContent>
   296→   296→            <SelectItem value="all">All Status</SelectItem>
   297→   297→            {QUOTATION_STATUSES.map((s) => <SelectItem key={s} value={s}>{s.charAt(0).toUpperCase() + s.slice(1)}</SelectItem>)}
   298→   298→          </SelectContent>
   299→   299→        </Select>
   300→   300→      </div>
   301→   301→
   302→   302→      {/* Content */}
   303→   303→      {loading ? (
   304→   304→        <div className="space-y-3">
   305→   305→          {Array.from({ length: 5 }).map((_, i) => <Skeleton key={i} className="h-16 rounded-xl" />)}
   306→   306→        </div>
   307→   307→      ) : quotations.length === 0 ? (
   308→   308→        <Card className="rounded-xl border-slate-200/60">
   309→   309→          <div className="flex flex-col items-center justify-center py-16 text-slate-400">
   310→   310→            <FileText className="w-10 h-10 mb-3" />
   311→   311→            <p className="text-sm font-medium">No quotations found</p>
   312→   312→            <p className="text-xs mt-1">Create your first quotation</p>
   313→   313→          </div>
   314→   314→        </Card>
   315→   315→      ) : (
   316→   316→        <>
   317→   317→          {/* Mobile Cards */}
   318→   318→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:hidden gap-4">
   319→   319→            {quotations.map((q) => {
   320→   320→              const qItems = parseItems(q.items);
   321→   321→              return (
   322→   322→                <motion.div key={q.id} initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }}>
   323→   323→                  <Card className="rounded-xl border-slate-200/60 shadow-sm cursor-pointer hover:border-emerald-200 transition-colors"
   324→   324→                    onClick={() => fetchQuotationDetail(q.id)}>
   325→   325→                    <CardContent className="p-4">
   326→   326→                      <div className="flex items-start justify-between">
   327→   327→                        <div>
   328→   328→                          <p className="text-sm font-mono font-bold text-slate-800">{q.quotationNumber}</p>
   329→   329→                          <p className="text-xs text-slate-500 mt-0.5">{q.lead?.company || q.lead?.name || '—'}</p>
   330→   330→                        </div>
   331→   331→                        <Badge className={`text-[11px] gap-1 ${STATUS_COLORS[q.status] || 'bg-slate-100 text-slate-600'} border-0`}>
   332→   332→                          {statusIcon(q.status)} {q.status}
   333→   333→                        </Badge>
   334→   334→                      </div>
   335→   335→                      <div className="mt-3 flex items-center justify-between">
   336→   336→                        <span className="text-xs text-slate-500">{formatDate(q.createdAt)}</span>
   337→   337→                        <span className="text-lg font-bold text-emerald-700">{formatAED(q.total)}</span>
   338→   338→                      </div>
   339→   339→                    </CardContent>
   340→   340→                  </Card>
   341→   341→                </motion.div>
   342→   342→              );
   343→   343→            })}
   344→   344→          </div>
   345→   345→
   346→   346→          {/* Desktop Table */}
   347→   347→          <Card className="rounded-xl border-slate-200/60 shadow-sm overflow-hidden hidden lg:block">
   348→   348→            <Table>
   349→   349→              <TableHeader>
   350→   350→                <TableRow className="bg-slate-50/80">
   351→   351→                  <TableHead className="text-xs uppercase tracking-wide text-slate-500">Quotation #</TableHead>
   352→   352→                  <TableHead className="text-xs uppercase tracking-wide text-slate-500">Lead / Company</TableHead>
   353→   353→                  <TableHead className="text-xs uppercase tracking-wide text-slate-500">Items</TableHead>
   354→   354→                  <TableHead className="text-xs uppercase tracking-wide text-slate-500">Subtotal</TableHead>
   355→   355→                  <TableHead className="text-xs uppercase tracking-wide text-slate-500">VAT</TableHead>
   356→   356→                  <TableHead className="text-xs uppercase tracking-wide text-slate-500">Total</TableHead>
   357→   357→                  <TableHead className="text-xs uppercase tracking-wide text-slate-500">Status</TableHead>
   358→   358→                  <TableHead className="text-xs uppercase tracking-wide text-slate-500">Date</TableHead>
   359→   359→                  <TableHead className="text-xs uppercase tracking-wide text-slate-500">Actions</TableHead>
   360→   360→                </TableRow>
   361→   361→              </TableHeader>
   362→   362→              <TableBody>
   363→   363→                {quotations.map((q) => {
   364→   364→                  const qItems = parseItems(q.items);
   365→   365→                  return (
   366→   366→                    <TableRow key={q.id} className="hover:bg-slate-50/50">
   367→   367→                      <TableCell className="font-mono text-sm font-semibold">{q.quotationNumber}</TableCell>
   368→   368→                      <TableCell className="text-sm text-slate-600">
   369→   369→                        {q.lead?.company || q.lead?.name || '—'}
   370→   370→                      </TableCell>
   371→   371→                      <TableCell className="text-sm text-slate-500">{qItems.length} item{qItems.length !== 1 ? 's' : ''}</TableCell>
   372→   372→                      <TableCell className="text-sm text-slate-600">{formatAED(q.subtotal)}</TableCell>
   373→   373→                      <TableCell className="text-sm text-slate-600">{formatAED(q.tax)}</TableCell>
   374→   374→                      <TableCell className="text-sm font-bold text-emerald-700">{formatAED(q.total)}</TableCell>
   375→   375→                      <TableCell>
   376→   376→                        <Badge className={`text-[11px] gap-1 ${STATUS_COLORS[q.status] || 'bg-slate-100 text-slate-600'} border-0`}>
   377→   377→                          {statusIcon(q.status)} {q.status}
   378→   378→                        </Badge>
   379→   379→                      </TableCell>
   380→   380→                      <TableCell className="text-xs text-slate-500">{formatDate(q.createdAt)}</TableCell>
   381→   381→                      <TableCell>
   382→   382→                        <div className="flex items-center gap-1">
   383→   383→                          <Button variant="ghost" size="sm" className="h-7 px-2 text-xs" onClick={() => fetchQuotationDetail(q.id)}>
   384→   384→                            <Eye className="w-3.5 h-3.5 mr-1" /> View
   385→   385→                          </Button>
   386→   386→                          {q.status === 'draft' && (
   387→   387→                            <Button variant="ghost" size="sm" className="h-7 px-2 text-xs text-emerald-600"
   388→   388→                              onClick={() => handleStatusUpdate(q.id, 'sent')}>
   389→   389→                              <Send className="w-3.5 h-3.5 mr-1" /> Send
   390→   390→                            </Button>
   391→   391→                          )}
