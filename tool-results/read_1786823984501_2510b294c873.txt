    13→    13→async function buildFleetContext(organizationId: string | null) {
    14→    14→  const orgFilter: { organizationId?: string } = organizationId
    15→    15→    ? { organizationId }
    16→    16→    : {};
    17→    17→
    18→    18→  const todayStart = new Date();
    19→    19→  todayStart.setHours(0, 0, 0, 0);
    20→    20→  const todayEnd = new Date();

    22→    22→
    23→    23→  const [
    24→    24→    vehicleCount,
    25→    25→    activeVehicles,
    26→    26→    inactiveVehicles,
    27→    27→    maintenanceVehicles,
    28→    28→    driverCount,

    30→    30→    openMaintenanceCount,
    31→    31→    todayTrips,
    32→    32→    todayTripsWithDistance,
    33→    33→    topDrivers,
    34→    34→    recentAlerts,
    35→    35→    upcomingMaintenance,
    36→    36→    vehicleTypeBreakdown,
    37→    37→  ] = await Promise.all([
    38→    38→    db.vehicle.count({ where: orgFilter }),
    39→    39→    db.vehicle.count({ where: { ...orgFilter, status: 'active' } }),
    40→    40→    db.vehicle.count({ where: { ...orgFilter, status: 'inactive' } }),
    41→    41→    db.vehicle.count({ where: { ...orgFilter, status: 'maintenance' } }),
    42→    42→    db.driver.count({ where: orgFilter }),
    43→    43→    db.alert.count({ where: { ...orgFilter, status: 'open' } }),
    44→    44→    db.maintenanceRecord.count({
    45→    45→      where: { ...orgFilter, status: { in: ['upcoming', 'overdue', 'in_progress'] } },
    46→    46→    }),
    47→    47→    db.trip.count({
    48→    48→      where: {
    49→    49→        vehicle: orgFilter ? { organizationId: orgFilter.organizationId } : undefined,
    50→    50→        startTime: { gte: todayStart, lte: todayEnd },
    51→    51→      },
    52→    52→    }),
    53→    53→    db.trip.aggregate({
    54→    54→      _sum: { distance: true, duration: true },
    55→    55→      where: {
    56→    56→        vehicle: orgFilter ? { organizationId: orgFilter.organizationId } : undefined,
    57→    57→        startTime: { gte: todayStart, lte: todayEnd },
    58→    58→      },
    59→    59→    }),
    60→    60→    db.driver.findMany({
    61→    61→      where: orgFilter,
    62→    62→      orderBy: { score: 'desc' },
    63→    63→      take: 5,
    64→    64→      select: { name: true, score: true, totalTrips: true, totalDistance: true, totalViolations: true, status: true },
    65→    65→    }),
    66→    66→    db.alert.findMany({
    67→    67→      where: { ...orgFilter, status: 'open' },
    68→    68→      orderBy: { createdAt: 'desc' },
    69→    69→      take: 10,
    70→    70→      select: { type: true, severity: true, vehiclePlate: true, driverName: true, message: true, createdAt: true },
    71→    71→    }),
    72→    72→    db.maintenanceRecord.findMany({
    73→    73→      where: { ...orgFilter, status: { in: ['upcoming', 'overdue'] } },
    74→    74→      orderBy: { scheduledDate: 'asc' },
    75→    75→      take: 10,
    76→    76→      include: { vehicle: { select: { plateNumber: true, make: true, model: true } } },
    77→    77→    }),
    78→    78→    db.vehicle.groupBy({
    79→    79→      by: ['vehicleType'],
    80→    80→      where: orgFilter,
    81→    81→      _count: { vehicleType: true },
    82→    82→    }),
    83→    83→  ]);
    84→    84→
    85→    85→  const totalDistanceToday = todayTripsWithDistance._sum?.distance ?? 0;
    86→    86→  const totalDurationToday = todayTripsWithDistance._sum?.duration ?? 0;
    87→    87→
    88→    88→  // Total mileage across all vehicles
    89→    89→  const totalMileageResult = await db.vehicle.aggregate({
    90→    90→    _sum: { mileage: true },
    91→    91→    where: orgFilter,
    92→    92→  });
    93→    93→  const totalMileage = totalMileageResult._sum.mileage ?? 0;
    94→    94→
    95→    95→  return {
    96→    96→    vehicleCount,
    97→    97→    activeVehicles,
    98→    98→    inactiveVehicles,
    99→    99→    maintenanceVehicles,
   100→   100→    driverCount,
   101→   101→    openAlertsCount,
   102→   102→    openMaintenanceCount,
   103→   103→    todayTrips,
   104→   104→    totalDistanceToday: Math.round(totalDistanceToday),
   105→   105→    totalDurationToday,
   106→   106→    totalMileage: Math.round(totalMileage),
   107→   107→    topDrivers,
   108→   108→    recentAlerts,
   109→   109→    upcomingMaintenance,
   110→   110→    vehicleTypeBreakdown: vehicleTypeBreakdown.map((v) => ({
   111→   111→      type: v.vehicleType ?? 'Unspecified',
   112→   112→      count: v._count.vehicleType,
   113→   113→    })),
   114→   114→  };
   115→   115→}

   526→   526→    };
   527→   527→
   528→   528→    if (!message || typeof message !== 'string' || message.trim().length === 0) {
   529→   529→      return NextResponse.json(
   530→   530→        { error: 'Message is required' },

   532→   532→      );

   541→   541→        where: { id: conversationId },

   605→   605→// GET /api/ai/chat — List recent conversations
   606→   606→// ────────────────────────────────────────────────
   607→   607→export async function GET(request: Request) {
   608→   608→  try {
   609→   609→    const { user, error } = await getAuthUser(request);
   610→   610→    if (error) return error;
   611→   611→
   612→   612→    const conversations = await db.aIConversation.findMany({
   613→   613→      where: {
   614→   614→        OR: [
   615→   615→          { userId: user.id },
   616→   616→          ...(user.organizationId ? [{ organizationId: user.organizationId }] : []),
   617→   617→        ],
   618→   618→      },
   619→   619→      orderBy: { updatedAt: 'desc' },
   620→   620→      take: 50,
   621→   621→      select: {
   622→   622→        id: true,
   623→   623→        type: true,
   624→   624→        createdAt: true,
   625→   625→        updatedAt: true,
   626→   626→        messages: true,
   627→   627→      },
   628→   628→    });
   629→   629→
   630→   630→    // Parse first user message as title, and message count
   631→   631→    const enriched = conversations.map((c) => {
   632→   632→      let parsed: ChatMessage[] = [];
   633→   633→      try {
   634→   634→        parsed = JSON.parse(c.messages) as ChatMessage[];
   635→   635→      } catch {
   636→   636→        // ignore
   637→   637→      }
   638→   638→
   639→   639→      const firstUserMsg = parsed.find((m) => m.role === 'user');
   640→   640→      const title = firstUserMsg
   641→   641→        ? firstUserMsg.content.slice(0, 80) + (firstUserMsg.content.length > 80 ? '…' : '')
   642→   642→        : 'New conversation';
   643→   643→
   644→   644→      return {
   645→   645→        id: c.id,
   646→   646→        type: c.type,
   647→   647→        title,
   648→   648→        messageCount: parsed.length,
   649→   649→        createdAt: c.createdAt,
   650→   650→        updatedAt: c.updatedAt,
   651→   651→      };
   652→   652→    });
   653→   653→
   654→   654→    return NextResponse.json({ conversations: enriched });
   655→   655→  } catch (err) {
   656→   656→    console.error('AI Conversations list error:', err);
   657→   657→    return NextResponse.json(
   658→   658→      { error: 'Internal server error' },
   659→   659→      { status: 500 },
   660→   660→    );
   661→   661→  }
