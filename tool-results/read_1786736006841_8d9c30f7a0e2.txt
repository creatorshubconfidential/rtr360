   361→function authFetch(url: string, options: RequestInit = {}) {
   362→  const token = typeof window !== 'undefined' ? localStorage.getItem('rtr_token') : null;
   363→  return fetch(url, {
   364→    ...options,
   365→    headers: {
   366→      'Content-Type': 'application/json',
   367→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
   368→      ...options.headers,
   369→    },
   370→  });
   371→}

  1856→export default function Home() {

  1862→      try {
  1863→        const token = localStorage.getItem('rtr_token');
  1864→        if (!token) {
  1865→          return;
  1866→        }
  1867→        const res = await fetch('/api/auth/me', {
  1868→          headers: { Authorization: `Bearer ${token}` },
  1869→        });

  1871→        const data = await res.json();

  1873→      } catch {
  1874→        localStorage.removeItem('rtr_token');
  1875→        localStorage.removeItem('rtr_user');
  1876→      } finally {
  1877→        setLoading(false);
  1878→      }
  1879→    };
  1880→
  1881→    checkAuth();
  1882→  }, []);
  1883→
  1884→  const handleLogin = (loggedInUser: UserSession, _token: string) => {
  1885→    setUser(loggedInUser);
  1886→  };
  1887→
  1888→  const handleLogout = () => {
  1889→    const token = localStorage.getItem('rtr_token');
  1890→    if (token) {
  1891→      fetch('/api/auth/logout', {
  1892→        method: 'POST',
  1893→        headers: { Authorization: `Bearer ${token}` },
  1894→      }).catch(() => {});
  1895→    }
  1896→    localStorage.removeItem('rtr_token');
  1897→    localStorage.removeItem('rtr_user');
  1898→    setUser(null);
  1899→    toast.info('You have been logged out');

  1901→
  1902→  if (loading) {
  1903→    return (
  1904→      <div className="min-h-screen flex items-center justify-center bg-[var(--rtr-bg)]">

  1910→        </div>
  1911→      </div>
  1912→    );
  1913→  }
  1914→
  1915→  if (!user) {
