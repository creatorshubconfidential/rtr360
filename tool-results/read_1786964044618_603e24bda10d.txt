   823→      { key: 'dashboard_refresh_interval', value: '30', type: 'number' },
   824→    ];
   825→
   826→    for (const s of settingsData) {
   827→      await db.setting.upsert({
   828→        where: { key: s.key },
   829→        update: { value: s.value },
   830→        create: { key: s.key, value: s.value, type: s.type },
   831→      });
   832→    }
   833→    results.settings = settingsData.length;
   834→
   835→    logger.info('Demo data seeded successfully', results);
   836→
   837→    return NextResponse.json({
   838→      message: 'Demo data seeded successfully',
   839→      seeded: true,
   840→      schemaSync: schemaSyncResult,
   841→      results,
   842→    });
   843→  } catch (error) {
   844→    logger.error('Demo seed error', { error });
   845→    return NextResponse.json(
   846→      { error: 'Demo seed failed', details: String(error) },
   847→      { status: 500 }
   848→    );
   849→  }
   850→}
