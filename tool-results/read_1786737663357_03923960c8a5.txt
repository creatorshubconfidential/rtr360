   939→function LeadsView() {
   940→  const [leads, setLeads] = useState<Lead[]>([]);
   941→  const [totalPages, setTotalPages] = useState(1);
   942→  const [page, setPage] = useState(1);
   943→  const [search, setSearch] = useState('');
   944→  const [statusFilter, setStatusFilter] = useState<string>('all');
   945→  const [priorityFilter, setPriorityFilter] = useState<string>('all');
   946→  const [loading, setLoading] = useState(true);
   947→  const [createOpen, setCreateOpen] = useState(false);
   948→  const [submitting, setSubmitting] = useState(false);
   949→
   950→  // Pipeline counts
   951→  const [pipelineCounts, setPipelineCounts] = useState<Record<string, number>>({});
   952→
   953→  // Create form
   954→  const [form, setForm] = useState({
   955→    name: '',
   956→    email: '',
   957→    phone: '',
   958→    company: '',
   959→    emirate: '',
   960→    vehicleCount: '',
   961→    vehicleType: '',
   962→    requirement: '',
   963→    source: '',
   964→    priority: 'medium',
   965→  });
   966→
   967→  const fetchLeads = useCallback(async () => {
   968→    setLoading(true);
   969→    try {
   970→      const params = new URLSearchParams({ page: String(page), limit: '10' });
   971→      if (search) params.set('search', search);
   972→      if (statusFilter !== 'all') params.set('status', statusFilter);
   973→      if (priorityFilter !== 'all') params.set('priority', priorityFilter);
   974→
   975→      const res = await authFetch(`/api/leads?${params.toString()}`);
   976→      const data = await res.json();
   977→      if (res.ok) {
   978→        setLeads(data.leads || []);
   979→        setTotalPages(data.pagination?.totalPages || 1);
   980→
   981→        // Calculate pipeline counts
   982→        const counts: Record<string, number> = {};
   983→        (data.leads || []).forEach((l: Lead) => {
   984→          counts[l.status] = (counts[l.status] || 0) + 1;
   985→        });
   986→        setPipelineCounts(counts);
   987→      }
   988→    } catch {
   989→      toast.error('Failed to load leads');
   990→    } finally {
   991→      setLoading(false);
   992→    }
   993→  }, [page, search, statusFilter, priorityFilter]);
   994→
   995→  useEffect(() => {
   996→    fetchLeads();
   997→  }, [fetchLeads]);
   998→
   999→  const handleCreate = async () => {
  1000→    if (!form.name.trim()) {
  1001→      toast.error('Lead name is required');
  1002→      return;
  1003→    }
  1004→    setSubmitting(true);
  1005→    try {
  1006→      const res = await authFetch('/api/leads', {
  1007→        method: 'POST',
  1008→        body: JSON.stringify({
  1009→          ...form,
  1010→          vehicleCount: form.vehicleCount ? parseInt(form.vehicleCount) : null,
  1011→        }),
  1012→      });
  1013→      const data = await res.json();
  1014→      if (!res.ok) {
  1015→        toast.error(data.error || 'Failed to create lead');
  1016→        return;
  1017→      }
  1018→      toast.success('Lead created successfully');
  1019→      setCreateOpen(false);
  1020→      setForm({ name: '', email: '', phone: '', company: '', emirate: '', vehicleCount: '', vehicleType: '', requirement: '', source: '', priority: 'medium' });
  1021→      fetchLeads();
  1022→    } catch {
  1023→      toast.error('Failed to create lead');
  1024→    } finally {
  1025→      setSubmitting(false);
  1026→    }
  1027→  };
  1028→
  1029→  const handleStatusUpdate = async (leadId: string, newStatus: string) => {
  1030→    try {
  1031→      const res = await authFetch(`/api/leads/${leadId}`, {
  1032→        method: 'PATCH',
  1033→        body: JSON.stringify({ status: newStatus }),
  1034→      });
  1035→      if (!res.ok) {
  1036→        const data = await res.json();
  1037→        toast.error(data.error || 'Failed to update status');
  1038→        return;
  1039→      }
  1040→      toast.success(`Lead moved to ${newStatus}`);
