     1→'use client';
     2→
     3→import { useState, useEffect, useCallback } from 'react';
     4→import { toast } from 'sonner';
     5→import { motion } from 'framer-motion';
     6→import {
     7→  Plus,
     8→  Search,
     9→  ChevronLeft,
    10→  ChevronRight,
    11→  Eye,
    12→  Send,
    13→  FileText,
    14→  CheckCircle,
    15→  XCircle,
    16→  Clock,
    17→} from 'lucide-react';
    18→import { Button } from '@/components/ui/button';
    19→import { Card, CardContent } from '@/components/ui/card';
    20→import { Badge } from '@/components/ui/badge';
    21→import {
    22→  Dialog,
    23→  DialogContent,
    24→  DialogHeader,
    25→  DialogTitle,
    26→  DialogFooter,
    27→} from '@/components/ui/dialog';
    28→import { Input } from '@/components/ui/input';
    29→import { Label } from '@/components/ui/label';
    30→import { Skeleton } from '@/components/ui/skeleton';
    31→import { Textarea } from '@/components/ui/textarea';
    32→import {
    33→  Select,
    34→  SelectContent,
    35→  SelectItem,
    36→  SelectTrigger,
    37→  SelectValue,
    38→} from '@/components/ui/select';
    39→import {
    40→  Table,
    41→  TableBody,
    42→  TableCell,
    43→  TableHead,
    44→  TableHeader,
    45→  TableRow,
    46→} from '@/components/ui/table';
    47→import { authFetch, formatAED, formatDate, formatDateTime } from '@/lib/api';
    48→import { STATUS_COLORS, QUOTATION_STATUSES, DEFAULT_QUOTATION_ITEMS, QUOTATION_TERMS } from '@/lib/constants';
    49→import type { Quotation, QuotationItem } from '@/lib/types';
    50→
    51→export default function QuotationsView() {
    52→  const [quotations, setQuotations] = useState<Quotation[]>([]);
    53→  const [totalPages, setTotalPages] = useState(1);
    54→  const [total, setTotal] = useState(0);
    55→  const [page, setPage] = useState(1);
    56→  const [search, setSearch] = useState('');
    57→  const [statusFilter, setStatusFilter] = useState<string>('all');
    58→  const [loading, setLoading] = useState(true);
    59→  const [selectedQuotation, setSelectedQuotation] = useState<Quotation | null>(null);
    60→  const [detailOpen, setDetailOpen] = useState(false);
    61→
    62→  // Create quotation
    63→  const [createOpen, setCreateOpen] = useState(false);
    64→  const [submitting, setSubmitting] = useState(false);
    65→  const [items, setItems] = useState<QuotationItem[]>([...DEFAULT_QUOTATION_ITEMS]);
    66→  const [leadSearch, setLeadSearch] = useState('');
    67→  const [leadResults, setLeadResults] = useState<Array<{ id: string; name: string; company: string | null }>>([]);
    68→  const [selectedLeadId, setSelectedLeadId] = useState<string | null>(null);
    69→  const [qNotes, setQNotes] = useState('');
    70→
    71→  const fetchQuotations = useCallback(async () => {
    72→    setLoading(true);
    73→    try {
    74→      const params = new URLSearchParams({ page: String(page), limit: '15' });
    75→      if (search) params.set('search', search);
    76→      if (statusFilter !== 'all') params.set('status', statusFilter);
    77→      const res = await authFetch(`/api/quotations?${params.toString()}`);
    78→      const data = await res.json();
    79→      if (res.ok) {
    80→        setQuotations(data.quotations || []);
    81→        setTotalPages(data.pagination?.totalPages || 1);
    82→        setTotal(data.pagination?.total || 0);
    83→      }
    84→    } catch { toast.error('Failed to load quotations'); }
    85→    finally { setLoading(false); }
    86→  }, [page, search, statusFilter]);
    87→
    88→  useEffect(() => { fetchQuotations(); }, [fetchQuotations]);
    89→
    90→  // Search leads for quotation creation
    91→  useEffect(() => {
    92→    if (!leadSearch.trim()) { setLeadResults([]); return; }
    93→    const timer = setTimeout(async () => {
    94→      try {
    95→        const res = await authFetch(`/api/leads?search=${encodeURIComponent(leadSearch)}&limit=5`);
    96→        const data = await res.json();
    97→        if (res.ok) setLeadResults(data.leads || []);
    98→      } catch { /* ignore */ }
    99→    }, 300);
   100→    return () => clearTimeout(timer);
   101→  }, [leadSearch]);
   102→
   103→  const fetchQuotationDetail = async (id: string) => {
   104→    try {
   105→      const res = await authFetch(`/api/quotations/${id}`);
   106→      const data = await res.json();
   107→      if (res.ok) {
   108→        setSelectedQuotation(data.quotation);
   109→        setDetailOpen(true);
   110→      }
   111→    } catch { toast.error('Failed to load quotation'); }
   112→  };
   113→
   114→  const handleStatusUpdate = async (id: string, status: string) => {
   115→    try {
   116→      const res = await authFetch(`/api/quotations/${id}`, {
   117→        method: 'PATCH',
   118→        body: JSON.stringify({ status }),
   119→      });
   120→      if (res.ok) {
   121→        toast.success(`Quotation marked as ${status}`);
   122→        fetchQuotations();
   123→      }
   124→    } catch { toast.error('Failed to update'); }
   125→  };
   126→
   127→  const handleCreate = async () => {
   128→    if (items.length === 0) { toast.error('Add at least one item'); return; }
   129→    setSubmitting(true);
   130→    try {
   131→      const res = await authFetch('/api/quotations', {
   132→        method: 'POST',
   133→        body: JSON.stringify({
   134→          leadId: selectedLeadId,
   135→          items,
   136→          notes: qNotes || undefined,
   137→          terms: QUOTATION_TERMS,
   138→        }),
   139→      });
   140→      const data = await res.json();
   141→      if (res.ok) {
   142→        toast.success(`Quotation ${data.quotation.quotationNumber} created!`);
   143→        setCreateOpen(false);
   144→        setItems([...DEFAULT_QUOTATION_ITEMS]);
   145→        setQNotes('');
   146→        setSelectedLeadId(null);
   147→        setLeadSearch('');
   148→        fetchQuotations();
   149→      } else { toast.error(data.error || 'Failed to create'); }
   150→    } catch { toast.error('Failed to create quotation'); }
   151→    finally { setSubmitting(false); }
   152→  };
   153→
   154→  const subtotal = items.reduce((s, i) => s + (i.quantity * i.unitPrice), 0);
   155→  const tax = Math.round(subtotal * 5 / 100 * 100) / 100;
   156→  const totalVal = Math.round((subtotal + tax) * 100) / 100;
   157→
   158→  const statusIcon = (status: string) => {
   159→    switch (status) {
   160→      case 'draft': return <FileText className="w-3.5 h-3.5" />;
   161→      case 'sent': return <Send className="w-3.5 h-3.5" />;
   162→      case 'accepted': return <CheckCircle className="w-3.5 h-3.5" />;
   163→      case 'rejected': return <XCircle className="w-3.5 h-3.5" />;
   164→      case 'expired': return <Clock className="w-3.5 h-3.5" />;
   165→      default: return null;
   166→    }
   167→  };
   168→
   169→  // Parse items from JSON string
   170→  const parseItems = (itemsJson: string): QuotationItem[] => {
   171→    try { return JSON.parse(itemsJson); } catch { return []; }
   172→  };
   173→
   174→  return (
   175→    <div className="space-y-5">
   176→      {/* Header */}
   177→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   178→        <div className="flex items-center gap-3">
   179→          <div>
   180→            <h2 className="text-2xl font-bold text-slate-900">Quotations</h2>
   181→            <p className="text-sm text-slate-500">Create and manage customer quotations</p>
   182→          </div>
   183→          <Badge variant="secondary" className="bg-emerald-100 text-emerald-700 border-0 text-sm px-2.5">
   184→            {total}
   185→          </Badge>
   186→        </div>
   187→        <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   188→          <DialogTrigger asChild>
   189→            <Button className="bg-emerald-600 hover:bg-emerald-700 text-white gap-2">
   190→              <Plus className="w-4 h-4" /> New Quotation
   191→            </Button>
   192→          </DialogTrigger>
   193→          <DialogContent className="max-w-2xl max-h-[85vh] overflow-y-auto">
   194→            <DialogHeader><DialogTitle>Create New Quotation</DialogTitle></DialogHeader>
   195→            <div className="space-y-4 py-2">
   196→              {/* Lead Search */}
   197→              <div className="space-y-2">
   198→                <Label>Link to Lead (optional)</Label>
   199→                <Input
   200→                  placeholder="Search lead by name or company..."
   201→                  value={leadSearch}
   202→                  onChange={(e) => setLeadSearch(e.target.value)}
   203→                />
   204→                {leadResults.length > 0 && !selectedLeadId && (
   205→                  <div className="border border-slate-200 rounded-lg max-h-32 overflow-y-auto">
   206→                    {leadResults.map((l) => (
   207→                      <button
   208→                        key={l.id}
   209→                        className="w-full text-left px-3 py-2 hover:bg-slate-50 text-sm flex items-center justify-between"
   210→                        onClick={() => { setSelectedLeadId(l.id); setLeadSearch(l.name + (l.company ? ` - ${l.company}` : '')); setLeadResults([]); }}
   211→                      >
   212→                        <span className="font-medium text-slate-800">{l.name}</span>
   213→                        <span className="text-xs text-slate-400">{l.company || ''}</span>
   214→                      </button>
   215→                    ))}
   216→                  </div>
   217→                )}
   218→                {selectedLeadId && (
   219→                  <div className="flex items-center gap-2 text-sm text-emerald-600">
   220→                    <CheckCircle className="w-4 h-4" />
   221→                    <span>Linked to: {leadSearch}</span>
   222→                    <button className="text-slate-400 hover:text-red-500 ml-auto" onClick={() => { setSelectedLeadId(null); setLeadSearch(''); }}>
   223→                      ✕
   224→                    </button>
   225→                  </div>
   226→                )}
   227→              </div>
   228→
   229→              {/* Line Items */}
   230→              <div>
   231→                <Label className="text-sm font-semibold">Line Items</Label>
   232→                <div className="mt-2 space-y-2">
   233→                  {items.map((item, idx) => (
   234→                    <div key={idx} className="flex items-center gap-2">
   235→                      <Input className="flex-1 h-9 text-sm" placeholder="Description" value={item.description}
   236→                        onChange={(e) => { const u = [...items]; u[idx] = { ...u[idx], description: e.target.value }; setItems(u); }} />
   237→                      <Input className="w-20 h-9 text-sm text-center" type="number" placeholder="Qty" value={item.quantity}
   238→                        onChange={(e) => { const u = [...items]; u[idx] = { ...u[idx], quantity: parseInt(e.target.value) || 0 }; setItems(u); }} />
   239→                      <Input className="w-28 h-9 text-sm text-right" type="number" placeholder="AED" value={item.unitPrice}
   240→                        onChange={(e) => { const u = [...items]; u[idx] = { ...u[idx], unitPrice: parseFloat(e.target.value) || 0 }; setItems(u); }} />
   241→                      <p className="text-sm font-medium text-slate-700 w-24 text-right">
   242→                        {formatAED(item.quantity * item.unitPrice)}
   243→                      </p>
   244→                      <button className="p-1.5 text-red-400 hover:text-red-600 hover:bg-red-50 rounded-lg transition-colors"
   245→                        onClick={() => setItems(items.filter((_, i) => i !== idx))}>✕</button>
   246→                    </div>
   247→                  ))}
   248→                </div>
   249→                <Button variant="outline" size="sm" className="mt-2 gap-1.5"
   250→                  onClick={() => setItems([...items, { description: '', quantity: 1, unitPrice: 0 }])}>
   251→                  <Plus className="w-3.5 h-3.5" /> Add Item
   252→                </Button>
   253→              </div>
   254→
   255→              <div className="space-y-2">
   256→                <Label>Notes</Label>
   257→                <Textarea value={qNotes} onChange={(e) => setQNotes(e.target.value)} placeholder="Additional notes..." rows={2} />
   258→              </div>
   259→
   260→              {/* Totals */}
   261→              <div className="p-4 bg-slate-50 rounded-xl space-y-2">
   262→                <div className="flex justify-between text-sm">
   263→                  <span className="text-slate-500">Subtotal</span>
   264→                  <span className="font-medium">{formatAED(subtotal)}</span>
   265→                </div>
   266→                <div className="flex justify-between text-sm">
   267→                  <span className="text-slate-500">VAT (5%)</span>
   268→                  <span className="font-medium">{formatAED(tax)}</span>
   269→                </div>
   270→                <div className="flex justify-between text-base font-bold border-t border-slate-200 pt-2">
   271→                  <span>Total</span>
   272→                  <span className="text-emerald-700">{formatAED(totalVal)}</span>
   273→                </div>
   274→              </div>
   275→            </div>
   276→            <DialogFooter>
   277→              <Button variant="outline" onClick={() => setCreateOpen(false)}>Cancel</Button>
   278→              <Button className="bg-emerald-600 hover:bg-emerald-700 text-white" onClick={handleCreate} disabled={submitting}>
   279→                {submitting ? 'Creating...' : 'Create Quotation'}
   280→              </Button>
   281→            </DialogFooter>
   282→          </DialogContent>
   283→        </Dialog>
   284→      </div>
   285→
   286→      {/* Filters */}
   287→      <div className="flex flex-col sm:flex-row gap-3">
   288→        <div className="relative flex-1">
   289→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
   290→          <Input placeholder="Search quotation number..." value={search}
   291→            onChange={(e) => { setSearch(e.target.value); setPage(1); }} className="pl-10" />
   292→        </div>
   293→        <Select value={statusFilter} onValueChange={(v) => { setStatusFilter(v); setPage(1); }}>
   294→          <SelectTrigger className="w-full sm:w-44"><SelectValue placeholder="All Status" /></SelectTrigger>
   295→          <SelectContent>
   296→            <SelectItem value="all">All Status</SelectItem>
   297→            {QUOTATION_STATUSES.map((s) => <SelectItem key={s} value={s}>{s.charAt(0).toUpperCase() + s.slice(1)}</SelectItem>)}
   298→          </SelectContent>
   299→        </Select>
   300→      </div>
   301→
   302→      {/* Content */}
   303→      {loading ? (
   304→        <div className="space-y-3">
   305→          {Array.from({ length: 5 }).map((_, i) => <Skeleton key={i} className="h-16 rounded-xl" />)}
   306→        </div>
   307→      ) : quotations.length === 0 ? (
   308→        <Card className="rounded-xl border-slate-200/60">
   309→          <div className="flex flex-col items-center justify-center py-16 text-slate-400">
   310→            <FileText className="w-10 h-10 mb-3" />
   311→            <p className="text-sm font-medium">No quotations found</p>
   312→            <p className="text-xs mt-1">Create your first quotation</p>
   313→          </div>
   314→        </Card>
   315→      ) : (
   316→        <>
   317→          {/* Mobile Cards */}
   318→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:hidden gap-4">
   319→            {quotations.map((q) => {
   320→              const qItems = parseItems(q.items);
   321→              return (
   322→                <motion.div key={q.id} initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }}>
   323→                  <Card className="rounded-xl border-slate-200/60 shadow-sm cursor-pointer hover:border-emerald-200 transition-colors"
   324→                    onClick={() => fetchQuotationDetail(q.id)}>
   325→                    <CardContent className="p-4">
   326→                      <div className="flex items-start justify-between">
   327→                        <div>
   328→                          <p className="text-sm font-mono font-bold text-slate-800">{q.quotationNumber}</p>
   329→                          <p className="text-xs text-slate-500 mt-0.5">{q.lead?.company || q.lead?.name || '—'}</p>
   330→                        </div>
   331→                        <Badge className={`text-[11px] gap-1 ${STATUS_COLORS[q.status] || 'bg-slate-100 text-slate-600'} border-0`}>
   332→                          {statusIcon(q.status)} {q.status}
   333→                        </Badge>
   334→                      </div>
   335→                      <div className="mt-3 flex items-center justify-between">
   336→                        <span className="text-xs text-slate-500">{formatDate(q.createdAt)}</span>
   337→                        <span className="text-lg font-bold text-emerald-700">{formatAED(q.total)}</span>
   338→                      </div>
   339→                    </CardContent>
   340→                  </Card>
   341→                </motion.div>
   342→              );
   343→            })}
   344→          </div>
   345→
   346→          {/* Desktop Table */}
   347→          <Card className="rounded-xl border-slate-200/60 shadow-sm overflow-hidden hidden lg:block">
   348→            <Table>
   349→              <TableHeader>
   350→                <TableRow className="bg-slate-50/80">
   351→                  <TableHead className="text-xs uppercase tracking-wide text-slate-500">Quotation #</TableHead>
   352→                  <TableHead className="text-xs uppercase tracking-wide text-slate-500">Lead / Company</TableHead>
   353→                  <TableHead className="text-xs uppercase tracking-wide text-slate-500">Items</TableHead>
   354→                  <TableHead className="text-xs uppercase tracking-wide text-slate-500">Subtotal</TableHead>
   355→                  <TableHead className="text-xs uppercase tracking-wide text-slate-500">VAT</TableHead>
   356→                  <TableHead className="text-xs uppercase tracking-wide text-slate-500">Total</TableHead>
   357→                  <TableHead className="text-xs uppercase tracking-wide text-slate-500">Status</TableHead>
   358→                  <TableHead className="text-xs uppercase tracking-wide text-slate-500">Date</TableHead>
   359→                  <TableHead className="text-xs uppercase tracking-wide text-slate-500">Actions</TableHead>
   360→                </TableRow>
   361→              </TableHeader>
   362→              <TableBody>
   363→                {quotations.map((q) => {
   364→                  const qItems = parseItems(q.items);
   365→                  return (
   366→                    <TableRow key={q.id} className="hover:bg-slate-50/50">
   367→                      <TableCell className="font-mono text-sm font-semibold">{q.quotationNumber}</TableCell>
   368→                      <TableCell className="text-sm text-slate-600">
   369→                        {q.lead?.company || q.lead?.name || '—'}
   370→                      </TableCell>
   371→                      <TableCell className="text-sm text-slate-500">{qItems.length} item{qItems.length !== 1 ? 's' : ''}</TableCell>
   372→                      <TableCell className="text-sm text-slate-600">{formatAED(q.subtotal)}</TableCell>
   373→                      <TableCell className="text-sm text-slate-600">{formatAED(q.tax)}</TableCell>
   374→                      <TableCell className="text-sm font-bold text-emerald-700">{formatAED(q.total)}</TableCell>
   375→                      <TableCell>
   376→                        <Badge className={`text-[11px] gap-1 ${STATUS_COLORS[q.status] || 'bg-slate-100 text-slate-600'} border-0`}>
   377→                          {statusIcon(q.status)} {q.status}
   378→                        </Badge>
   379→                      </TableCell>
   380→                      <TableCell className="text-xs text-slate-500">{formatDate(q.createdAt)}</TableCell>
   381→                      <TableCell>
   382→                        <div className="flex items-center gap-1">
   383→                          <Button variant="ghost" size="sm" className="h-7 px-2 text-xs" onClick={() => fetchQuotationDetail(q.id)}>
   384→                            <Eye className="w-3.5 h-3.5 mr-1" /> View
   385→                          </Button>
   386→                          {q.status === 'draft' && (
   387→                            <Button variant="ghost" size="sm" className="h-7 px-2 text-xs text-emerald-600"
   388→                              onClick={() => handleStatusUpdate(q.id, 'sent')}>
   389→                              <Send className="w-3.5 h-3.5 mr-1" /> Send
   390→                            </Button>
   391→                          )}
   392→                          {(q.status === 'sent' || q.status === 'draft') && (
   393→                            <Button variant="ghost" size="sm" className="h-7 px-2 text-xs text-emerald-600"
   394→                              onClick={() => handleStatusUpdate(q.id, 'accepted')}>
   395→                              <CheckCircle className="w-3.5 h-3.5 mr-1" /> Accept
   396→                            </Button>
   397→                          )}
   398→                        </div>
   399→                      </TableCell>
   400→                    </TableRow>
   401→                  );
   402→                })}
   403→              </TableBody>
   404→            </Table>
   405→          </Card>
   406→
   407→          {/* Pagination */}
   408→          <div className="flex items-center justify-between px-4 py-3">
   409→            <p className="text-sm text-slate-500">Page {page} of {totalPages}</p>
   410→            <div className="flex gap-2">
   411→              <Button variant="outline" size="sm" onClick={() => setPage((p) => Math.max(1, p - 1))} disabled={page <= 1}>
   412→                <ChevronLeft className="w-4 h-4 mr-1" /> Previous
   413→              </Button>
   414→              <Button variant="outline" size="sm" onClick={() => setPage((p) => Math.min(totalPages, p + 1))} disabled={page >= totalPages}>
   415→                Next <ChevronRight className="w-4 h-4 ml-1" />
   416→              </Button>
   417→            </div>
   418→          </div>
   419→        </>
   420→      )}
   421→
   422→      {/* ─── Quotation Detail Dialog ─── */}
   423→      <Dialog open={detailOpen} onOpenChange={setDetailOpen}>
   424→        <DialogContent className="max-w-2xl max-h-[85vh] overflow-y-auto">
   425→          <DialogHeader>
   426→            <DialogTitle className="flex items-center justify-between">
   427→              <span>{selectedQuotation?.quotationNumber}</span>
   428→              {selectedQuotation && (
   429→                <Badge className={`${STATUS_COLORS[selectedQuotation.status] || 'bg-slate-100 text-slate-600'} border-0 gap-1`}>
   430→                  {statusIcon(selectedQuotation.status)} {selectedQuotation.status}
   431→                </Badge>
   432→              )}
   433→            </DialogTitle>
   434→          </DialogHeader>
   435→          {selectedQuotation && (() => {
   436→            const qItems = parseItems(selectedQuotation.items);
   437→            return (
   438→              <div className="space-y-6 py-2">
   439→                {/* Company Info */}
   440→                <div className="grid grid-cols-2 gap-4 p-4 bg-slate-50 rounded-xl">
   441→                  <div>
   442→                    <p className="text-xs text-slate-500 uppercase tracking-wide">Customer</p>
   443→                    <p className="text-sm font-semibold text-slate-800 mt-0.5">
   444→                      {selectedQuotation.lead?.company || selectedQuotation.lead?.name || 'N/A'}
   445→                    </p>
   446→                    {selectedQuotation.lead?.name && selectedQuotation.lead?.company && (
   447→                      <p className="text-xs text-slate-500">{selectedQuotation.lead.name}</p>
   448→                    )}
   449→                  </div>
   450→                  <div>
   451→                    <p className="text-xs text-slate-500 uppercase tracking-wide">RTR 360 Platform</p>
   452→                    <p className="text-sm font-semibold text-slate-800 mt-0.5">Powered by Mianx.ai</p>
   453→                    <p className="text-xs text-slate-500">Dubai, UAE</p>
   454→                  </div>
   455→                </div>
   456→
   457→                {/* Line Items Table */}
   458→                <div>
   459→                  <h4 className="text-sm font-semibold text-slate-700 mb-2">Line Items</h4>
   460→                  <Table>
   461→                    <TableHeader>
   462→                      <TableRow className="bg-slate-50">
