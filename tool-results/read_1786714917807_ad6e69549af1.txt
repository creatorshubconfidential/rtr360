   276→function authFetch(url: string, options: RequestInit = {}) {
   277→  const token = typeof window !== 'undefined' ? localStorage.getItem('rtr_token') : null;
   278→  return fetch(url, {
   279→    ...options,
   280→    headers: {
   281→      'Content-Type': 'application/json',
   282→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
   283→      ...options.headers,
   284→    },
   285→  });
   286→}

   297→  const handleLogin = async (e: React.FormEvent) => {
   298→    e.preventDefault();
   299→    if (!email || !password) {
   300→      toast.error('Please enter email and password');
   301→      return;
   302→    }
   303→    setLoading(true);
   304→    try {
   305→      const res = await fetch('/api/auth/login', {
   306→        method: 'POST',
   307→        headers: { 'Content-Type': 'application/json' },
   308→        body: JSON.stringify({ email, password }),
   309→      });

   311→      if (!res.ok) {
   312→        toast.error(data.error || 'Login failed');

   316→      localStorage.setItem('rtr_user', JSON.stringify(data.user));

   520→function DashboardView() {
   521→  const [stats, setStats] = useState<DashboardStats | null>(null);
   522→  const [leads, setLeads] = useState<Lead[]>([]);
   523→  const [alerts, setAlerts] = useState<Alert[]>([]);
   524→  const [loading, setLoading] = useState(true);
   525→
   526→  useEffect(() => {
   527→    const fetchData = async () => {
   528→      try {
   529→        const [statsRes, leadsRes, alertsRes] = await Promise.all([
   530→          authFetch('/api/dashboard/stats'),
   531→          authFetch('/api/leads?limit=5'),
   532→          authFetch('/api/dashboard/alerts?limit=5'),
   533→        ]);
   534→        const statsData = await statsRes.json();
   535→        if (statsRes.ok) setStats(statsData);
   536→
   537→        const leadsData = await leadsRes.json();
   538→        if (leadsRes.ok) setLeads(leadsData.leads || []);
   539→
   540→        if (alertsRes.ok) {
   541→          const alertsData = await alertsRes.json();
   542→          setAlerts(alertsData.alerts || []);
   543→        }
   544→      } catch {
   545→        // Silently fail for alerts — the endpoint may not exist
   546→      } finally {
   547→        setLoading(false);
   548→      }
   549→    };
   550→    fetchData();
   551→  }, []);
   552→
   553→  const kpiCards = stats
   554→    ? [
   555→        { icon: Truck, label: 'Total Vehicles', value: stats.totalVehicles, change: '+12%', color: 'bg-emerald-100 text-emerald-600' },
   556→        { icon: Activity, label: 'Active Vehicles', value: stats.activeVehicles, change: '+8%', color: 'bg-green-100 text-green-600' },
   557→        { icon: Users, label: 'Total Drivers', value: stats.totalDrivers, change: '+5%', color: 'bg-teal-100 text-teal-600' },
   558→        { icon: UserPlus, label: 'Open Leads', value: stats.totalLeads, change: '+23%', color: 'bg-amber-100 text-amber-600' },
   559→        { icon: AlertTriangle, label: 'Open Alerts', value: stats.openAlerts, change: '-5%', color: 'bg-red-100 text-red-600' },
   560→        { icon: Ticket, label: 'Open Tickets', value: stats.openTickets, change: '+2%', color: 'bg-orange-100 text-orange-600' },
   561→        { icon: Route, label: "Today's Trips", value: stats.todayTrips, change: '+15%', color: 'bg-purple-100 text-purple-600' },
   562→        { icon: Gauge, label: 'Total Distance (km)', value: stats.totalDistance.toLocaleString(), change: '+9%', color: 'bg-cyan-100 text-cyan-600' },
   563→      ]
   564→    : [];
   565→
   566→  const timeAgo = (dateStr: string) => {
   567→    const diff = Date.now() - new Date(dateStr).getTime();
   568→    const mins = Math.floor(diff / 60000);
   569→    if (mins < 60) return `${mins}m ago`;
   570→    const hrs = Math.floor(mins / 60);
   571→    if (hrs < 24) return `${hrs}h ago`;
   572→    const days = Math.floor(hrs / 24);
   573→    return `${days}d ago`;
   574→  };
   575→
   576→  if (loading) {
   577→    return (
   578→      <div className="space-y-6">
   579→        <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   580→          {Array.from({ length: 8 }).map((_, i) => (
   581→            <Skeleton key={i} className="h-28 rounded-xl" />
   582→          ))}
   583→        </div>

   585→          <Skeleton className="h-80 rounded-xl" />

  1542→export default function Home() {
  1543→  const [user, setUser] = useState<UserSession | null>(null);
  1544→  const [loading, setLoading] = useState(true);
  1545→
  1546→  useEffect(() => {
  1547→    const checkAuth = async () => {
  1548→      try {
  1549→        const token = localStorage.getItem('rtr_token');
  1550→        if (!token) {
  1551→          return;
  1552→        }
  1553→        const res = await fetch('/api/auth/me', {
  1554→          headers: { Authorization: `Bearer ${token}` },
  1555→        });
  1556→        if (!res.ok) throw new Error('Invalid token');
  1557→        const data = await res.json();
  1558→        setUser(data.user);
  1559→      } catch {
  1560→        localStorage.removeItem('rtr_token');
  1561→        localStorage.removeItem('rtr_user');
  1562→      } finally {
  1563→        setLoading(false);
  1564→      }
  1565→    };
  1566→
  1567→    checkAuth();
  1568→  }, []);
  1569→
  1570→  const handleLogin = (loggedInUser: UserSession, _token: string) => {
  1571→    setUser(loggedInUser);
  1572→  };
  1573→
  1574→  const handleLogout = () => {
  1575→    const token = localStorage.getItem('rtr_token');
  1576→    if (token) {
  1577→      fetch('/api/auth/logout', {
  1578→        method: 'POST',
  1579→        headers: { Authorization: `Bearer ${token}` },
  1580→      }).catch(() => {});
  1581→    }
  1582→    localStorage.removeItem('rtr_token');
  1583→    localStorage.removeItem('rtr_user');
  1584→    setUser(null);
  1585→    toast.info('You have been logged out');
  1586→  };
  1587→
  1588→  if (loading) {
  1589→    return (
  1590→      <div className="min-h-screen flex items-center justify-center bg-[var(--rtr-bg)]">
  1591→        <div className="flex flex-col items-center gap-4">
  1592→          <div className="w-12 h-12 bg-emerald-600 rounded-xl flex items-center justify-center">
  1593→            <Truck className="w-7 h-7 text-white" />
  1594→          </div>
  1595→          <div className="w-8 h-8 border-3 border-emerald-200 border-t-emerald-600 rounded-full animate-spin" />
  1596→        </div>
  1597→      </div>
  1598→    );
  1599→  }
  1600→
  1601→  if (!user) {
  1602→    return <LoginScreen onLogin={handleLogin} />;
  1603→  }
  1604→
  1605→  return <AdminDashboard user={user} onLogout={handleLogout} />;
  1606→}
