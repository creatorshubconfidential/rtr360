   355→function authFetch(url: string, options: RequestInit = {}) {
   356→  const token = typeof window !== 'undefined' ? localStorage.getItem('rtr_token') : null;
   357→  return fetch(url, {
   358→    ...options,
   359→    headers: {
   360→      'Content-Type': 'application/json',
   361→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
   362→      ...options.headers,
   363→    },
   364→  });
   365→}

   371→function LoginScreen({ onLogin }: { onLogin: (user: UserSession, token: string) => void }) {
   372→  const [email, setEmail] = useState('');
   373→  const [password, setPassword] = useState('');
   374→  const [loading, setLoading] = useState(false);
   375→
   376→  const handleLogin = async (e: React.FormEvent) => {
   377→    e.preventDefault();
   378→    if (!email || !password) {
   379→      toast.error('Please enter email and password');
   380→      return;
   381→    }
   382→    setLoading(true);
   383→    try {
   384→      const res = await fetch('/api/auth/login', {
   385→        method: 'POST',
   386→        headers: { 'Content-Type': 'application/json' },
   387→        body: JSON.stringify({ email, password }),
   388→      });
   389→      const data = await res.json();
   390→      if (!res.ok) {
   391→        toast.error(data.error || 'Login failed');
   392→        return;
   393→      }
   394→      localStorage.setItem('rtr_token', data.token);
   395→      localStorage.setItem('rtr_user', JSON.stringify(data.user));
   396→      onLogin(data.user, data.token);
   397→      toast.success('Welcome back!');
   398→    } catch {
   399→      toast.error('Network error. Please try again.');
   400→    } finally {
   401→      setLoading(false);
   402→    }
   403→  };
   404→
   405→  return (
   406→    <div className="min-h-screen flex">
   407→      {/* Decorative Left Panel */}
   408→      <div className="hidden lg:flex lg:w-1/2 bg-slate-900 relative overflow-hidden flex-col justify-between p-12">

   813→function LeadsView() {

   837→    source: '',
   838→    priority: 'medium',

   841→  const fetchLeads = useCallback(async () => {
   842→    setLoading(true);
   843→    try {
   844→      const params = new URLSearchParams({ page: String(page), limit: '10' });
   845→      if (search) params.set('search', search);
   846→      if (statusFilter !== 'all') params.set('status', statusFilter);
   847→      if (priorityFilter !== 'all') params.set('priority', priorityFilter);

   857→        (data.leads || []).forEach((l: Lead) => {
   858→          counts[l.status] = (counts[l.status] || 0) + 1;
   859→        });
   860→        setPipelineCounts(counts);

   862→    } catch {

   865→      setLoading(false);

   881→        method: 'POST',

   883→          ...form,
   884→          vehicleCount: form.vehicleCount ? parseInt(form.vehicleCount) : null,
