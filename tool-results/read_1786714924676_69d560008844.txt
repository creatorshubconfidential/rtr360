   276→   276→function authFetch(url: string, options: RequestInit = {}) {
   277→   277→  const token = typeof window !== 'undefined' ? localStorage.getItem('rtr_token') : null;
   278→   278→  return fetch(url, {
   279→   279→    ...options,
   280→   280→    headers: {
   281→   281→      'Content-Type': 'application/json',
   282→   282→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
   283→   283→      ...options.headers,
   284→   284→    },
   285→   285→  });
   286→   286→}

   292→   292→function LoginScreen({ onLogin }: { onLogin: (user: UserSession, token: string) => void }) {
   293→   293→  const [email, setEmail] = useState('');
   294→   294→  const [password, setPassword] = useState('');
   295→   295→  const [loading, setLoading] = useState(false);
   296→   296→
   297→   297→  const handleLogin = async (e: React.FormEvent) => {
   298→   298→    e.preventDefault();
   299→   299→    if (!email || !password) {
   300→   300→      toast.error('Please enter email and password');
   301→   301→      return;
   302→   302→    }
   303→   303→    setLoading(true);
   304→   304→    try {
   305→   305→      const res = await fetch('/api/auth/login', {
   306→   306→        method: 'POST',
   307→   307→        headers: { 'Content-Type': 'application/json' },
   308→   308→        body: JSON.stringify({ email, password }),
   309→   309→      });
   310→   310→      const data = await res.json();
   311→   311→      if (!res.ok) {
   312→   312→        toast.error(data.error || 'Login failed');
   313→   313→        return;
   314→   314→      }
   315→   315→      localStorage.setItem('rtr_token', data.token);
   316→   316→      localStorage.setItem('rtr_user', JSON.stringify(data.user));

   318→   318→      toast.success('Welcome back!');

   320→   320→      toast.error('Network error. Please try again.');
