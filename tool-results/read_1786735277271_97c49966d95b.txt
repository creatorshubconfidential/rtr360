   360→function authFetch(url: string, options: RequestInit = {}) {
   361→  const token = typeof window !== 'undefined' ? localStorage.getItem('rtr_token') : null;
   362→  return fetch(url, {
   363→    ...options,
   364→    headers: {
   365→      'Content-Type': 'application/json',
   366→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
   367→      ...options.headers,
   368→    },
   369→  });
   370→}

   376→function LoginScreen({ onLogin }: { onLogin: (user: UserSession, token: string) => void }) {
   377→  const [email, setEmail] = useState('');
   378→  const [password, setPassword] = useState('');
   379→  const [loading, setLoading] = useState(false);
   380→
   381→  const handleLogin = async (e: React.FormEvent) => {
   382→    e.preventDefault();
   383→    if (!email || !password) {
   384→      toast.error('Please enter email and password');
   385→      return;
   386→    }
   387→    setLoading(true);
   388→    try {
   389→      const res = await fetch('/api/auth/login', {
   390→        method: 'POST',
   391→        headers: { 'Content-Type': 'application/json' },
   392→        body: JSON.stringify({ email, password }),
   393→      });
   394→      const data = await res.json();
   395→      if (!res.ok) {
   396→        toast.error(data.error || 'Login failed');
   397→        return;
   398→      }
   399→      localStorage.setItem('rtr_token', data.token);
   400→      localStorage.setItem('rtr_user', JSON.stringify(data.user));
   401→      onLogin(data.user, data.token);
   402→      toast.success('Welcome back!');
   403→    } catch {
   404→      toast.error('Network error. Please try again.');
   405→    } finally {
   406→      setLoading(false);
   407→    }
   408→  };
   409→
   410→  return (
   411→    <div className="min-h-screen flex">
   412→      {/* Decorative Left Panel */}
   413→      <div className="hidden lg:flex lg:w-1/2 bg-slate-900 relative overflow-hidden flex-col justify-between p-12">

   818→function LeadsView() {

   824→  const [priorityFilter, setPriorityFilter] = useState<string>('all');
   825→  const [loading, setLoading] = useState(true);
   826→  const [createOpen, setCreateOpen] = useState(false);
   827→  const [submitting, setSubmitting] = useState(false);
   828→
   829→  // Pipeline counts
   830→  const [pipelineCounts, setPipelineCounts] = useState<Record<string, number>>({});
   831→
   832→  // Create form

   834→    name: '',

   836→    phone: '',
   837→    company: '',
   838→    emirate: '',
   839→    vehicleCount: '',
   840→    vehicleType: '',
   841→    requirement: '',
   842→    source: '',
   843→    priority: 'medium',
   844→  });
   845→
   846→  const fetchLeads = useCallback(async () => {
   847→    setLoading(true);
   848→    try {
   849→      const params = new URLSearchParams({ page: String(page), limit: '10' });
   850→      if (search) params.set('search', search);
   851→      if (statusFilter !== 'all') params.set('status', statusFilter);
   852→      if (priorityFilter !== 'all') params.set('priority', priorityFilter);

   854→      const res = await authFetch(`/api/leads?${params.toString()}`);

   856→      if (res.ok) {
   857→        setLeads(data.leads || []);
   858→        setTotalPages(data.pagination?.totalPages || 1);
   859→
   860→        // Calculate pipeline counts
   861→        const counts: Record<string, number> = {};
   862→        (data.leads || []).forEach((l: Lead) => {
   863→          counts[l.status] = (counts[l.status] || 0) + 1;
   864→        });
   865→        setPipelineCounts(counts);
   866→      }
   867→    } catch {
   868→      toast.error('Failed to load leads');
   869→    } finally {
   870→      setLoading(false);
   871→    }

   875→    fetchLeads();
   876→  }, [fetchLeads]);

   879→    if (!form.name.trim()) {
   880→      toast.error('Lead name is required');
   881→      return;
   882→    }
   883→    setSubmitting(true);
   884→    try {
   885→      const res = await authFetch('/api/leads', {
   886→        method: 'POST',
   887→        body: JSON.stringify({
   888→          ...form,
   889→          vehicleCount: form.vehicleCount ? parseInt(form.vehicleCount) : null,
   890→        }),
   891→      });

   893→      if (!res.ok) {
   894→        toast.error(data.error || 'Failed to create lead');
   895→        return;

   901→    } catch {
   902→      toast.error('Failed to create lead');
   903→    } finally {
   904→      setSubmitting(false);
   905→    }
   906→  };

   909→    try {
   910→      const res = await authFetch(`/api/leads/${leadId}`, {
   911→        method: 'PATCH',
   912→        body: JSON.stringify({ status: newStatus }),

   915→        const data = await res.json();
   916→        toast.error(data.error || 'Failed to update status');
   917→        return;
