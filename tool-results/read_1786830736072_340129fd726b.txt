   377→function authFetch(url: string, options: RequestInit = {}) {
   378→  const token = typeof window !== 'undefined' ? localStorage.getItem('rtr_token') : null;
   379→  return fetch(url, {
   380→    ...options,
   381→    headers: {
   382→      'Content-Type': 'application/json',
   383→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
   384→      ...options.headers,
   385→    },
   386→  });
   387→}

   393→function LoginScreen({ onLogin }: { onLogin: (user: UserSession, token: string) => void }) {
   394→  const [email, setEmail] = useState('');
   395→  const [password, setPassword] = useState('');
   396→  const [loading, setLoading] = useState(false);
   397→
   398→  const handleLogin = async (e: React.FormEvent) => {
   399→    e.preventDefault();
   400→    if (!email || !password) {
   401→      toast.error('Please enter email and password');
   402→      return;
   403→    }
   404→    setLoading(true);
   405→    try {
   406→      const res = await fetch('/api/auth/login', {
   407→        method: 'POST',
   408→        headers: { 'Content-Type': 'application/json' },
   409→        body: JSON.stringify({ email, password }),
   410→      });
   411→      const data = await res.json();
   412→      if (!res.ok) {
   413→        toast.error(data.error || 'Login failed');
   414→        return;
   415→      }
   416→      localStorage.setItem('rtr_token', data.token);
   417→      localStorage.setItem('rtr_user', JSON.stringify(data.user));
   418→      onLogin(data.user, data.token);
   419→      toast.success('Welcome back!');
   420→    } catch {
   421→      toast.error('Network error. Please try again.');
   422→    } finally {
   423→      setLoading(false);
   424→    }
   425→  };
