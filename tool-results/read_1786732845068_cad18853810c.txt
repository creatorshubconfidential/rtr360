   354→function authFetch(url: string, options: RequestInit = {}) {
   355→  const token = typeof window !== 'undefined' ? localStorage.getItem('rtr_token') : null;
   356→  return fetch(url, {
   357→    ...options,
   358→    headers: {
   359→      'Content-Type': 'application/json',
   360→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
   361→      ...options.headers,
   362→    },
   363→  });
   364→}

   370→function LoginScreen({ onLogin }: { onLogin: (user: UserSession, token: string) => void }) {
   371→  const [email, setEmail] = useState('');
   372→  const [password, setPassword] = useState('');
   373→  const [loading, setLoading] = useState(false);
   374→
   375→  const handleLogin = async (e: React.FormEvent) => {
   376→    e.preventDefault();
   377→    if (!email || !password) {
   378→      toast.error('Please enter email and password');
   379→      return;
   380→    }
   381→    setLoading(true);
   382→    try {
   383→      const res = await fetch('/api/auth/login', {
   384→        method: 'POST',
   385→        headers: { 'Content-Type': 'application/json' },
   386→        body: JSON.stringify({ email, password }),
   387→      });
   388→      const data = await res.json();
   389→      if (!res.ok) {
   390→        toast.error(data.error || 'Login failed');
   391→        return;
   392→      }
   393→      localStorage.setItem('rtr_token', data.token);
   394→      localStorage.setItem('rtr_user', JSON.stringify(data.user));
   395→      onLogin(data.user, data.token);
   396→      toast.success('Welcome back!');
   397→    } catch {
   398→      toast.error('Network error. Please try again.');
   399→    } finally {
   400→      setLoading(false);
   401→    }
   402→  };
   403→
   404→  return (
   405→    <div className="min-h-screen flex">
   406→      {/* Decorative Left Panel */}
   407→      <div className="hidden lg:flex lg:w-1/2 bg-slate-900 relative overflow-hidden flex-col justify-between p-12">

   812→function LeadsView() {

   817→  const [statusFilter, setStatusFilter] = useState<string>('all');
   818→  const [priorityFilter, setPriorityFilter] = useState<string>('all');
   819→  const [loading, setLoading] = useState(true);
   820→  const [createOpen, setCreateOpen] = useState(false);
   821→  const [submitting, setSubmitting] = useState(false);
   822→
   823→  // Pipeline counts
   824→  const [pipelineCounts, setPipelineCounts] = useState<Record<string, number>>({});
   825→
   826→  // Create form
   827→  const [form, setForm] = useState({
   828→    name: '',
   829→    email: '',
   830→    phone: '',
   831→    company: '',
   832→    emirate: '',
   833→    vehicleCount: '',
   834→    vehicleType: '',
   835→    requirement: '',
   836→    source: '',
   837→    priority: 'medium',
   838→  });
   839→
   840→  const fetchLeads = useCallback(async () => {
   841→    setLoading(true);
   842→    try {
   843→      const params = new URLSearchParams({ page: String(page), limit: '10' });
   844→      if (search) params.set('search', search);
   845→      if (statusFilter !== 'all') params.set('status', statusFilter);
   846→      if (priorityFilter !== 'all') params.set('priority', priorityFilter);
   847→
   848→      const res = await authFetch(`/api/leads?${params.toString()}`);
   849→      const data = await res.json();
   850→      if (res.ok) {
   851→        setLeads(data.leads || []);
   852→        setTotalPages(data.pagination?.totalPages || 1);
   853→
   854→        // Calculate pipeline counts
   855→        const counts: Record<string, number> = {};
   856→        (data.leads || []).forEach((l: Lead) => {
   857→          counts[l.status] = (counts[l.status] || 0) + 1;
   858→        });
   859→        setPipelineCounts(counts);
   860→      }
   861→    } catch {
   862→      toast.error('Failed to load leads');
   863→    } finally {
   864→      setLoading(false);
   865→    }
   866→  }, [page, search, statusFilter, priorityFilter]);

   868→  useEffect(() => {
   869→    fetchLeads();
   870→  }, [fetchLeads]);

   873→    if (!form.name.trim()) {
   874→      toast.error('Lead name is required');
   875→      return;
   876→    }
   877→    setSubmitting(true);
   878→    try {
   879→      const res = await authFetch('/api/leads', {
   880→        method: 'POST',
   881→        body: JSON.stringify({
   882→          ...form,
   883→          vehicleCount: form.vehicleCount ? parseInt(form.vehicleCount) : null,
   884→        }),
   885→      });

   888→        toast.error(data.error || 'Failed to create lead');
   889→        return;

   896→      toast.error('Failed to create lead');
   897→    } finally {
   898→      setSubmitting(false);

   904→      const res = await authFetch(`/api/leads/${leadId}`, {
   905→        method: 'PATCH',
   906→        body: JSON.stringify({ status: newStatus }),

   909→        const data = await res.json();
   910→        toast.error(data.error || 'Failed to update status');
   911→        return;
   912→      }
   913→      toast.success(`Lead moved to ${newStatus}`);

   916→      toast.error('Failed to update lead status');
