   347→function authFetch(url: string, options: RequestInit = {}) {
   348→  const token = typeof window !== 'undefined' ? localStorage.getItem('rtr_token') : null;
   349→  return fetch(url, {
   350→    ...options,
   351→    headers: {
   352→      'Content-Type': 'application/json',
   353→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
   354→      ...options.headers,
   355→    },
   356→  });
   357→}

   363→function LoginScreen({ onLogin }: { onLogin: (user: UserSession, token: string) => void }) {
   364→  const [email, setEmail] = useState('');
   365→  const [password, setPassword] = useState('');
   366→  const [loading, setLoading] = useState(false);
   367→
   368→  const handleLogin = async (e: React.FormEvent) => {
   369→    e.preventDefault();
   370→    if (!email || !password) {
   371→      toast.error('Please enter email and password');
   372→      return;
   373→    }
   374→    setLoading(true);
   375→    try {
   376→      const res = await fetch('/api/auth/login', {
   377→        method: 'POST',
   378→        headers: { 'Content-Type': 'application/json' },
   379→        body: JSON.stringify({ email, password }),
   380→      });
   381→      const data = await res.json();
   382→      if (!res.ok) {
   383→        toast.error(data.error || 'Login failed');
   384→        return;
   385→      }
   386→      localStorage.setItem('rtr_token', data.token);
   387→      localStorage.setItem('rtr_user', JSON.stringify(data.user));
   388→      onLogin(data.user, data.token);
   389→      toast.success('Welcome back!');
   390→    } catch {
   391→      toast.error('Network error. Please try again.');
   392→    } finally {
   393→      setLoading(false);
   394→    }
