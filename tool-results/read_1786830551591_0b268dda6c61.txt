    29→function authFetch(url: string, options: RequestInit = {}) {
    30→  const token = typeof window !== 'undefined' ? localStorage.getItem('rtr_token') : null;
    31→  return fetch(url, {
    32→    ...options,
    33→    headers: {
    34→      'Content-Type': 'application/json',
    35→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
    36→      ...options.headers,
    37→    },
    38→  });
    39→}

    94→function formatDuration(seconds: number): string {
    95→  if (!seconds || seconds <= 0) return '—';
    96→  if (seconds >= 3600) {
    97→    const h = Math.floor(seconds / 3600);
    98→    const m = Math.floor((seconds % 3600) / 60);
    99→    return `${h}h ${m}m`;
   100→  }
   101→  return `${Math.floor(seconds / 60)}m`;
   102→}
   103→
   104→function formatDateTime(v: string | null) {
   105→  if (!v) return '—';
   106→  return new Date(v).toLocaleDateString('en-AE', {
   107→    day: '2-digit', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit',
   108→  });
   109→}

   646→              </div>

   648→                <Label className="text-sm font-medium">Status</Label>
   649→                <Select value={form.status} onValueChange={(v) => setForm({ ...form, status: v })}>
   650→                  <SelectTrigger className="h-9"><SelectValue /></SelectTrigger>
   651→                  <SelectContent>

   653→                    <SelectItem value="completed">Completed</SelectItem>
   654→                    <SelectItem value="cancelled">Cancelled</SelectItem>

   662→                <Input
   663→                  type="number"
   664→                  placeholder="0"
   665→                  value={form.maxSpeed}

   672→                  type="number"
   673→                  placeholder="0"
   674→                  value={form.avgSpeed}
